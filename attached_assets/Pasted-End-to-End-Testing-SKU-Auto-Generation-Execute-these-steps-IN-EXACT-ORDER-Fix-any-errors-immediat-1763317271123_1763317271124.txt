app.get("/api/listings", async (_req, res) => {
  try {
    // Get all listings from database first
    const dbListings = await db
      .select()
      .from(listings)
      .leftJoin(inventoryItems, eq(listings.inventoryId, inventoryItems.inventoryId))
      .orderBy(desc(listings.lastSyncedAt));

    // Enrich with live eBay data for published listings
    const enrichedListings = await Promise.all(
      dbListings.map(async (row) => {
        // If listing has eBay offer ID, fetch live status
        if (row.listings.ebayOfferId) {
          try {
            const ebayOffer = await ebayClient.getOffer(row.listings.ebayOfferId);
            
            return {
              ...row.listings,
              inventory: row.inventory_items,
              ebayStatus: ebayOffer.status,
              ebayPrice: ebayOffer.pricingSummary?.price?.value,
              isLiveOnEbay: ebayOffer.status === 'PUBLISHED',
            };
          } catch (error) {
            console.warn(`[Listings] Could not fetch eBay data for ${row.listings.ebayOfferId}:`, error);
            // Return DB data if eBay fetch fails
            return {
              ...row.listings,
              inventory: row.inventory_items,
            };
          }
        }
        
        // No eBay offer yet - return DB data
        return {
          ...row.listings,
          inventory: row.inventory_items,
        };
      })
    );

    res.json(enrichedListings);
  } catch (e: any) {
    console.error("[Listings] Error:", e);
    res.status(500).json({ error: e.message });
  }
});
