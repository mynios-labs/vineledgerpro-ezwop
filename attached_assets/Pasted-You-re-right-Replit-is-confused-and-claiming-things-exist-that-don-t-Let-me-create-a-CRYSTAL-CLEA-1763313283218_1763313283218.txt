import { getAccessToken } from './ebay';

const EBAY_API_BASE = process.env.EBAY_ENV === "production" 
  ? "https://api.ebay.com"
  : "https://api.sandbox.ebay.com";

const MARKETPLACE_ID = "EBAY_US";

export interface EbayOffer {
  offerId: string;
  sku?: string | null;
  status: 'PUBLISHED' | 'UNPUBLISHED' | 'ENDED';
  availableQuantity?: number;
  pricingSummary?: { price?: { value: string; currency: string; }; };
  listing?: { listingId?: string; title?: string; };
}

interface RequestOptions {
  method?: 'GET' | 'POST' | 'PUT' | 'DELETE';
  body?: any;
  headers?: Record<string, string>;
  requiresMarketplace?: boolean;
}

export class EbayClient {
  private token: string | null = null;

  private async request<T>(endpoint: string, options: RequestOptions = {}): Promise<T> {
    const { method = 'GET', body, headers: customHeaders = {}, requiresMarketplace = true } = options;

    if (!this.token) {
      this.token = await getAccessToken();
    }

    const url = endpoint.startsWith('http') ? endpoint : `${EBAY_API_BASE}${endpoint}`;

    const headers: Record<string, string> = {
      'Authorization': `Bearer ${this.token}`,
      'Content-Type': 'application/json',
      'Accept': 'application/json',
      'Content-Language': 'en-US',
      'Accept-Language': 'en-US',
      ...customHeaders,
    };

    if (requiresMarketplace && !endpoint.includes('/identity/') && !endpoint.includes('/commerce/')) {
      headers['X-EBAY-C-MARKETPLACE-ID'] = MARKETPLACE_ID;
    }

    const fetchOptions: RequestInit = {
      method,
      headers,
      body: body ? JSON.stringify(body) : undefined,
    };

    console.log(`[EbayClient] ${method} ${endpoint}`);

    const response = await fetch(url, fetchOptions);
    const responseText = await response.text();

    if (!response.ok) {
      if (response.status === 429) {
        const retryAfter = response.headers.get('Retry-After');
        const delay = retryAfter ? parseInt(retryAfter) * 1000 : 2000;
        console.log(`[EbayClient] Rate limited, retrying after ${delay}ms`);
        await new Promise(resolve => setTimeout(resolve, delay));
        return this.request<T>(endpoint, options);
      }

      let errorData;
      try { errorData = JSON.parse(responseText); } 
      catch { errorData = { message: responseText }; }

      throw new Error(`eBay API error ${response.status}: ${JSON.stringify(errorData)}`);
    }

    return responseText ? JSON.parse(responseText) : null;
  }

  async *getAllOffers(): AsyncGenerator<EbayOffer[]> {
    let offset = 0;
    const limit = 200;
    let hasMore = true;

    while (hasMore) {
      const url = `/sell/inventory/v1/offer?marketplace_id=${MARKETPLACE_ID}&limit=${limit}&offset=${offset}`;
      const response: any = await this.request(url);
      const items = response.offers || [];

      if (items.length > 0) {
        yield items;
        offset += items.length;
        hasMore = items.length === limit;
      } else {
        hasMore = false;
      }
    }
  }

  async getOffer(offerId: string): Promise<EbayOffer> {
    return this.request<EbayOffer>(`/sell/inventory/v1/offer/${offerId}`);
  }

  async createOffer(offer: any): Promise<{ offerId: string }> {
    const result = await this.request<{ offerId: string }>('/sell/inventory/v1/offer', { method: 'POST', body: offer });
    if (!result.offerId) throw new Error('eBay create offer failed: No offerId in response');
    return result;
  }

  async publishOffer(offerId: string): Promise<{ listingId: string }> {
    const result = await this.request<{ listingId: string }>(`/sell/inventory/v1/offer/${offerId}/publish`, { method: 'POST' });
    if (!result.listingId) throw new Error('eBay publish offer failed: No listingId in response');
    return result;
  }

  async upsertInventoryItem(sku: string, item: any): Promise<void> {
    await this.request<void>(`/sell/inventory/v1/inventory_item/${encodeURIComponent(sku)}`, { method: 'PUT', body: item });
    console.log(`[EbayClient] Inventory item upserted: ${sku}`);
  }
}
