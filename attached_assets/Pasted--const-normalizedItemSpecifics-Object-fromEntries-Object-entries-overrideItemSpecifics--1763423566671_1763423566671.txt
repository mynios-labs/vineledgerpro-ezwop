        setMissingItemSpecifics(missingSpecifics);

        variant: "destructive",
      });

    }

    setPublishResult({
      error: error.message,
      details: error.details || [error.message],
      failedStep: error.failedStep,
      trace: error.trace,
    });
    toast({
      title: "Publish failed",
      description: error.message,
      variant: "destructive",
    });
  },
